# ─────────────────────────────────────────────────────────────────────────────
# All packages are pinned to exact versions to prevent pip's dependency
# resolver from hitting "resolution-too-deep" on the complex transitive graph
# of llama-index + langchain + ragas + datasets.
#
# Upgrade strategy: bump one package at a time and re-verify compatibility.
# ─────────────────────────────────────────────────────────────────────────────

# ── PDF parsing ───────────────────────────────────────────────────────────────
pymupdf==1.24.0

# ── Tokenization ─────────────────────────────────────────────────────────────
tiktoken==0.7.0

# ── Vector DB client ─────────────────────────────────────────────────────────
# 4.7.x satisfies llama-index-vector-stores-weaviate 1.x (requires >=4.5.7)
# and the Weaviate 1.27+ server we run in docker-compose.
weaviate-client==4.7.1

# ── HTTP client (explicit pin keeps weaviate-client and llama-index aligned) ──
httpx==0.27.0

# ── OpenAI SDK ───────────────────────────────────────────────────────────────
# 1.51 is compatible with llama-index-embeddings-openai 0.2.x and our API code.
openai==1.51.0

# ── Utilities ────────────────────────────────────────────────────────────────
python-dotenv==1.0.1

# ── API layer ─────────────────────────────────────────────────────────────────
fastapi==0.111.0
uvicorn[standard]==0.29.0

# ── LlamaIndex ────────────────────────────────────────────────────────────────
# 0.11.x era — all four packages must be from the same release era.
# llama-index-vector-stores-weaviate 1.1.x requires core>=0.11 and weaviate-client>=4.5.7.
llama-index-core==0.11.22
llama-index-vector-stores-weaviate==1.1.0
llama-index-readers-file==0.2.2
llama-index-embeddings-openai==0.2.5

# ── LangChain ────────────────────────────────────────────────────────────────
# 0.2.x family — must match ragas 0.2.x which requires langchain-core<0.3.0.
# core / openai / text-splitters must all be from the same 0.2.x era.
langchain-core==0.2.43
langchain-openai==0.1.25
langchain-text-splitters==0.2.4

# ── Evaluation ───────────────────────────────────────────────────────────────
# ragas 0.2.x requires langchain-core>=0.2.0,<0.3.0 and datasets 2.x.
ragas==0.2.6
datasets==2.21.0
numpy==1.26.4

# ── ONNX inference (fastembed) ────────────────────────────────────────────────
# fastembed wraps ONNX Runtime for 3-5x faster CPU inference vs PyTorch.
# Used as an optional backend for BGE-M3 embeddings (set USE_ONNX=false to fall back).
fastembed==0.4.1

# ── ONNX inference ────────────────────────────────────────────────────────────
# onnxruntime is used by _OnnxBgeEmbeddings in semantic_chunker.py.
# On first use the BGE-M3 model is exported to ~/.cache/bge_onnx/ via
# torch.onnx.export (one-time ~2 min); subsequent runs load from the ONNX cache
# and run via onnxruntime, giving ~1.4x faster CPU inference vs PyTorch.
onnxruntime==1.23.2

# ── Semantic chunking (Phase 3) ───────────────────────────────────────────────
# SemanticChunker lives in langchain_experimental.text_splitter
# HuggingFaceEmbeddings for BGE-M3 is in langchain_huggingface
# sentence-transformers is the underlying runtime for loading BGE-M3
# All three require langchain-core>=0.2.0,<0.3.0 — compatible with the pin above.
langchain-experimental==0.0.65
langchain-huggingface==0.0.3
sentence-transformers==3.0.1
# transformers is pinned to <4.50 because 4.50+ enforces torch>=2.6 for all .bin
# weight loading (CVE-2025-32434).  PyTorch 2.3+ dropped macOS x86_64 support so
# torch cannot be upgraded on Intel Macs; 4.47.x is the latest compatible release.
# sentence-transformers 3.0.1 requires transformers>=4.34.0,<5 — 4.47.0 satisfies that.
transformers==4.47.0
